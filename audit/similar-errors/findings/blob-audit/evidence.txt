export function addGood(save: SharedSaveData, good: RuntimeGood, count = good.num): void {
  if (Number.isSafeInteger(count) === false || count <= 0) return
  if (save.illustrated.includes(good.id) === false) save.illustrated.push(good.id)
  const list = save.selfGoodDatas[good.type]
  const index = inventoryIndex(save, good.type, good.id)
  const next = (index >= 0 ? Number(list[index].split('*')[1] ?? 0) : 0) + count
  const code = '_' + good.id + '*' + String(next)
  if (index >= 0) list[index] = code
  else list.push(code)
}

export function takeGood(save: SharedSaveData, type: GoodType, id: string, count = 1): boolean {
  const current = inventoryCount(save, type, id)
  if (current < count) return false
  const index = inventoryIndex(save, type, id)
  const list = save.selfGoodDatas[type]
  const left = current - count
  if (left > 0) list[index] = '_' + id + '*' + String(left)
  else list.splice(index, 1)
  return true
}

/**
 * Backfills canonical defaults added after an older bundle was written. A bundle carries the
 * canonical snapshot twice (`shared` and `engine.save`); both copies get the same default so the
 * bundle's shared/engine identity assertion still holds.
 */
function backfillPortableBundleDefaults(bundle: Record<string, unknown>): string[] {
  const filled = backfillLegacySharedPetInfo(bundle.shared)
  if (isRecord(bundle.engine)) filled.push(...backfillLegacySharedPetInfo(bundle.engine.save))
  return filled
}

/**
 * Canonical petInfo fields that predate older saves, with the value startSave() repairs them to:
 * a pet that never had the field gets the same 3 diamonds a new one starts with (#75).
 */
export const LEGACY_PET_INFO_DEFAULTS = { diamond: INITIAL_DIAMOND_COUNT } as const

/**
 * Backfills canonical petInfo fields that older saves/bundles predate, in place.
 * Only absent keys are filled: a present-but-invalid value still fails the shared schema,
 * so upgrading an old snapshot can never launder bad data. Returns the filled field names.
 */
export function backfillLegacySharedPetInfo(save: unknown): string[] {
  if (typeof save !== 'object' || save === null || Array.isArray(save)) return []
  const petInfo = (save as Record<string, unknown>).petInfo
  if (typeof petInfo !== 'object' || petInfo === null || Array.isArray(petInfo)) return []
  const record = petInfo as Record<string, unknown>
  const filled: string[] = []
  for (const [key, value] of Object.entries(LEGACY_PET_INFO_DEFAULTS)) {
    if (record[key] !== undefined) continue
    record[key] = value
    filled.push(key)
  }
  return filled
}
